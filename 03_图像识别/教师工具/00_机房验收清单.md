### 1.2 结构预检（标准库脚本，先在机房跑一遍）

把下面脚本存为 `机房验收_pptx.py`，放在 `Python基础/` 目录下，用 IDLE 或 `python 机房验收_pptx.py` 运行。**只用标准库，无需 python-pptx。**

```python
# -*- coding: utf-8 -*-
import pathlib, re, zipfile

root = pathlib.Path(".")
expect = {
    "第03讲_程序会算术": 16,
    "第04讲_小数之谜": 15,
    "第05讲_控制结构入门": 12,
}
for folder, pages in expect.items():
    f = root / folder / "课件.pptx"
    z = zipfile.ZipFile(str(f))
    slides = [n for n in z.namelist()
              if re.match(r"ppt/slides/slide\d+\.xml$", n)]
    notes = [n for n in z.namelist()
             if re.match(r"ppt/notesSlides/notesSlide\d+\.xml$", n)]
    body = "".join(z.read(n).decode("utf-8", "ignore") for n in slides)
    ok_page = len(slides) == pages
    ok_note = len(notes) == pages
    print("%s | 页数 %d/%d %s | 备注 %d/%d %s | 雅黑 %s | Consolas %s" % (
        folder, len(slides), pages, "OK" if ok_page else "FAIL",
        len(notes), pages, "OK" if ok_note else "FAIL",
        "有" if "Microsoft YaHei" in body else "无",
        "有" if "Consolas" in body else "无"))
```

预期输出：三行全部 `OK`，`雅黑 有`、`Consolas 有`。

### 2.2 全库编译 + 实机运行（标准库脚本）

把下面脚本存为 `机房验收_py.py`，放在 `Python基础/` 目录下运行。**关键：工作目录必须是讲次文件夹**，否则代码里的 `数据/xxx.txt` 相对路径会找不到文件。

```python
# -*- coding: utf-8 -*-
import pathlib, py_compile, subprocess, sys

root = pathlib.Path(".")
pys = sorted(p for p in root.rglob("*.py")
             if "代码" in p.parts
             and ".tmp" not in p.parts
             and "__pycache__" not in p.parts)
bad = []
for p in pys:
    try:
        py_compile.compile(str(p), doraise=True)
    except Exception as e:
        bad.append(("编译", str(p), str(e)[:80]))

ran = 0
for p in pys:
    lesson = p.parent.parent          # 讲次文件夹，作为 cwd
    r = subprocess.run([sys.executable, "代码/" + p.name],
                       cwd=str(lesson),
                       capture_output=True, text=True, encoding="utf-8")
    ran += 1
    if r.returncode != 0:
        bad.append(("运行", str(p), (r.stderr or "")[-120:]))
    elif p.name.startswith(("demo", "answer", "参考")):
        print("OK  %-46s -> %s" % (str(p), repr(r.stdout.strip()[:60])))

print("\n共 %d 个 .py；编译+运行异常 %d 项" % (ran, len(bad)))
for x in bad:
    print("  FAIL", x)
```

用下面命令核对 38 个 `.txt` 的规格（标准库，可直接粘进 IDLE）：

```python
import pathlib
TEXT_SAMPLES = {"形状卡片.txt"}   # 中文文本样例，不是 0/1 网格
for p in sorted(pathlib.Path(".").rglob("*.txt")):
    if ".tmp" in p.parts: continue
    b = p.read_bytes()
    crlf_ok = b.endswith(b"\r\n") and b"\n" not in b.replace(b"\r\n", b"")
    if p.name in TEXT_SAMPLES:
        try:
            b.decode("utf-8"); dec = True
        except UnicodeDecodeError:
            dec = False
        ok = dec and b[:3] != b"\xef\xbb\xbf" and crlf_ok
    else:
        ok = (b[:3] != b"\xef\xbb\xbf" and set(b) <= set(b"01\r\n") and crlf_ok)
    print(("OK  " if ok else "FAIL"), len(b), p)
```

预期：38 行全部 `OK`。
